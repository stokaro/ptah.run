  • # データベース自身のスキーマをファイルに書き出す。$ ptah db read --db-url sqlite://app.db | tee schema.sqlCREATE TABLE "users" (  "id" INTEGER PRIMARY KEY,  "email" TEXT NOT NULL); # ドリフト検査は、そのファイルと取得元のデータベースを比べる。$ ptah schema drift --schema-file schema.sql \    --db-url sqlite://app.dbNo schema drift detected. # 目標スキーマを書き換えて、列を 1 つ要求する。$ cat > schema.sql <<'SQL'CREATE TABLE "users" (  "id" INTEGER PRIMARY KEY,  "email" TEXT NOT NULL,  "created_at" TEXT);SQL # 今度は同じ検査が報告を返す。$ ptah schema drift --schema-file schema.sql \    --db-url sqlite://app.dbSchema drift detected (highest severity: warning).Failure threshold: all. Failing: true.Database: sqlite://app.db Findings:- columns_added: 1 (warning) # 差を埋めるのに何が要るかを尋ねる。ドライランは何も実行しない。$ ptah schema apply --schema-file schema.sql \    --db-url sqlite://app.db --dry-runPlanned schema changes:ALTER TABLE "users" ADD COLUMN "created_at" TEXT; # レビュー済みの計画を実行する。使い捨てのファイルなら --auto-approve でよい。$ ptah schema apply --schema-file schema.sql \    --db-url sqlite://app.db --auto-approvePlanned schema changes:ALTER TABLE "users" ADD COLUMN "created_at" TEXT;Auto-approval enabled; applying schema changes.Schema apply completed successfully. # そして同じ検査を、3 度目に。$ ptah schema drift --schema-file schema.sql \    --db-url sqlite://app.dbNo schema drift detected.

  • # スキーマは Go の構造体。宣言はアノテーションのほう。$ cat entities/user.gopackage entities //ptah:schema:table name="users"type User struct {	//ptah:schema:field name="id" type="SERIAL" primary="true"	ID int64	//ptah:schema:field name="email" type="VARCHAR(255)" not_null="true"	Email string	//ptah:schema:field name="plan" type="ENUM" enum="free,team,enterprise"	Plan string} # PostgreSQL 向けに出力する。$ ptah schema render --root-dir . --dialect postgres-- Statement 1/2CREATE TYPE "enum_user_plan" AS ENUM ('free', 'team', 'enterprise'); -- Statement 2/2-- POSTGRES TABLE: users --CREATE TABLE "users" (  "id" SERIAL PRIMARY KEY NOT NULL,  "email" VARCHAR(255) NOT NULL,  "plan" enum_user_plan); # 同じファイル、違うのはフラグ 1 つ。$ ptah schema render --root-dir . --dialect sqlite-- Statement 1/1CREATE TABLE "users" (  "id" INTEGER PRIMARY KEY,  "email" TEXT NOT NULL,  "plan" TEXT CHECK (plan IN ('free', 'team', 'enterprise'))); # SQLite に enum 型はないので CHECK になる。落ちたものは何もない。
